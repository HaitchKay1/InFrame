cmake_minimum_required(VERSION 3.28...3.30)
option(INFRAME_CORE_ONLY "Build camera-motion tests without OBS/OpenCV" OFF)
if(INFRAME_CORE_ONLY)
  project(inframe-tests LANGUAGES CXX)
  enable_testing()
  add_executable(tracker-tests tests/tracker-tests.cpp)
  target_compile_features(tracker-tests PRIVATE cxx_std_17)
  add_test(NAME tracker COMMAND tracker-tests)
  return()
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version} LANGUAGES C CXX)
option(ENABLE_FRONTEND_API "Use OBS frontend API" OFF)
option(ENABLE_QT "Use Qt" OFF)
include(compilerconfig)
include(defaults)
include(helpers)
find_package(libobs REQUIRED)
find_package(OpenCV 4.8 REQUIRED COMPONENTS core imgproc objdetect dnn)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/data/models/face_detection_yunet_2023mar.onnx" _model_hash)
if(NOT _model_hash STREQUAL "8f2383e4dd3cfbb4553ea8718107fc0423210dc964f9f4280604804ed2552fa4")
  message(FATAL_ERROR "Bundled YuNet model is missing or has an unexpected checksum")
endif()
find_package(Threads REQUIRED)
add_library(${CMAKE_PROJECT_NAME} MODULE src/plugin.cpp src/detector.cpp)
target_compile_features(${CMAKE_PROJECT_NAME} PRIVATE cxx_std_17)
target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE OBS::libobs ${OpenCV_LIBS} Threads::Threads)
if(MSVC)
  target_compile_options(${CMAKE_PROJECT_NAME} PRIVATE /utf-8)
  target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
set_target_properties_plugin(${CMAKE_PROJECT_NAME} PROPERTIES OUTPUT_NAME ${_name})
include(CTest)
if(BUILD_TESTING)
  add_executable(tracker-tests tests/tracker-tests.cpp)
  target_compile_features(tracker-tests PRIVATE cxx_std_17)
  add_test(NAME tracker COMMAND tracker-tests)
  add_executable(detector-tests tests/detector-tests.cpp src/detector.cpp)
  target_compile_features(detector-tests PRIVATE cxx_std_17)
  target_include_directories(detector-tests PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(detector-tests PRIVATE ${OpenCV_LIBS} Threads::Threads)
  add_test(NAME detector COMMAND detector-tests "${CMAKE_CURRENT_SOURCE_DIR}/data/models/face_detection_yunet_2023mar.onnx")
endif()
